// Create a record
await fetch('https://reqres.in/app/collections/todos/records', {
  method: 'POST',
  headers: {
    'Authorization': 'Bearer <session_token>',
    'Content-Type': 'application/json'
  },
  body: JSON.stringify({
    data: { title: 'Launch MVP', completed: false }
  })
})

// Read it back
