            function methodClass(m) { return 'rp-method rp-m-' + m; }
            function escapeHtml(s) { return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); }

            function highlightJson(obj) {
                var json = escapeHtml(JSON.stringify(obj, null, 2));
                return json.replace(/("(\\u[a-fA-F0-9]{4}|\\[^u]|[^\\"])*"(\s*:)?|\b(true|false)\b|\bnull\b|-?\d+(?:\.\d*)?(?:[eE][+\-]?\d+)?)/g, function (m) {
                    var cls = 'rp-num';
                    if (/^"/.test(m)) { cls = /:$/.test(m) ? 'rp-key' : 'rp-str'; }
                    else if (/true|false/.test(m)) { cls = 'rp-bool'; }
                    else if (/null/.test(m)) { cls = 'rp-null'; }
                    return '<span class="' + cls + '">' + m + '</span>';
                });
            }

            function curlFor(p) {
                var url = API_ORIGIN + p.path;
                var parts = ['curl'];
                if (p.method !== 'GET') parts.push('-X ' + p.method);
                parts.push('"' + url + '"');
                if (p.body) {
                    parts.push("-H \"Content-Type: application/json\"");
                    parts.push("-d '" + JSON.stringify(p.body) + "'");
                }
                return parts.join(' \\\n  ');
            }

            function selectPreset(p) {
                active = p;
                elMethod.textContent = p.method;
                elMethod.className = methodClass(p.method);
                elUrl.textContent = p.path;
                if (p.body) {
                    elBodyWrap.classList.remove('rp-hidden');
                    elBody.value = JSON.stringify(p.body, null, 2);
                } else {
                    elBodyWrap.classList.add('rp-hidden');
                    elBody.value = '';
                }
                elCurl.textContent = curlFor(p);
                Array.prototype.forEach.call(elPresets.querySelectorAll('.rp-preset'), function (c) {
                    c.setAttribute('data-active', c.getAttribute('data-id') === p.id ? 'true' : 'false');
                });
            }

            var lastGroup = null;
            presets.forEach(function (p) {
                if (p.group !== lastGroup) {
                    lastGroup = p.group;
                    var h = document.createElement('p');
                    h.className = 'rp-group';
                    h.textContent = p.group;
                    elPresets.appendChild(h);
                }
                var btn = document.createElement('button');
                btn.type = 'button';
                btn.className = 'rp-preset';
                btn.setAttribute('data-id', p.id);
                btn.setAttribute('data-active', p === active ? 'true' : 'false');
                btn.innerHTML =
                    '<span class="' + methodClass(p.method) + '">' + p.method + '</span>' +
                    '<span class="rp-preset-label">' + p.label + '</span>' +
                    (p.note ? '<span class="rp-note">' + p.note + '</span>' : '');
                btn.addEventListener('click', function () {
                    selectPreset(p);
                    track('demo_endpoint_selected', { endpoint: p.id, method: p.method, path: p.path });
                });
                elPresets.appendChild(btn);
            });
            selectPreset(active);

            elCopy.addEventListener('click', function () {
                var text = elCurl.textContent;
                var done = function () { elCopy.textContent = 'Copied'; elCopy.classList.add('rp-copied'); setTimeout(function () { elCopy.textContent = 'Copy'; elCopy.classList.remove('rp-copied'); }, 1400); };
                if (navigator.clipboard && navigator.clipboard.writeText) { navigator.clipboard.writeText(text).then(done, function () {}); }
                else { try { var ta = document.createElement('textarea'); ta.value = text; document.body.appendChild(ta); ta.select(); document.execCommand('copy'); document.body.removeChild(ta); done(); } catch (e) {} }
                track('demo_curl_copied', { endpoint: active.id });
            });

            function setStatus(code, statusText) {
                elStatus.className = 'rp-status-' + Math.floor(code / 100);
                elStatus.textContent = code + ' ' + (statusText || '');
            }
            function busy(on) { elSend.disabled = on; elSpin.classList.toggle('rp-hidden', !on); }

            function openModal() {
                elModal.classList.remove('rp-hidden');
                track('demo_signup_modal_shown', { sent_count: sentCount });
            }
            function lockSend() {
                gated = true;
                var lbl = document.getElementById('rp-send-label');
                if (lbl) lbl.textContent = 'Sign up to keep going';
            }
            function closeModal() { elModal.classList.add('rp-hidden'); }

            // Fire a view event once the playground actually enters the viewport.
            (function () {
                var fired = false;
                function fire() { if (fired) return; fired = true; track('demo_playground_viewed', {}); }
                if (window.IntersectionObserver && elSection) {
                    var io = new IntersectionObserver(function (entries) {
                        entries.forEach(function (en) { if (en.isIntersecting) { fire(); io.disconnect(); } });
                    }, { threshold: 0.25 });
                    io.observe(elSection);
                } else { fire(); }
            })();

            elSend.addEventListener('click', function () {
                if (gated) { openModal(); return; }
                elNudge.classList.add('rp-hidden');
                busy(true);
                elStatus.className = 'rp-status';
                elStatus.textContent = 'Sending...';
                elTime.textContent = '';

                var opts = { method: active.method, headers: {} };
                if (active.body) { opts.headers['Content-Type'] = 'application/json'; opts.body = elBody.value; }
                var t0 = (window.performance && performance.now) ? performance.now() : Date.now();

                fetch(active.path, opts).then(function (res) {
                    var ms = Math.max(1, Math.round(((window.performance && performance.now) ? performance.now() : Date.now()) - t0));
                    elTime.textContent = ms + ' ms';
                    setStatus(res.status, res.statusText);

                    var remaining = res.headers.get('x-ratelimit-remaining');
                    var limit = res.headers.get('x-ratelimit-limit');
                    if (remaining !== null && limit !== null) { elQuota.textContent = remaining + ' / ' + limit + ' demo requests left today'; }

                    return res.text().then(function (text) {
                        var parsed = null, ok = false;
                        try { parsed = JSON.parse(text); ok = true; } catch (e) {}
                        if (ok) { elOut.innerHTML = highlightJson(parsed); }
                        else if (text) { elOut.textContent = text; }
                        else { elOut.textContent = '// ' + res.status + ' ' + (res.statusText || '') + ' (empty response body)'; }

                        track('demo_request_sent', {
                            endpoint: active.id, method: active.method, path: active.path,
                            status: res.status, duration_ms: ms, request_number: sentCount + 1
                        });
                        // Reuse the site's canonical playground-usage event so this
                        // stitches with the historical legacy_console_used series.
                        try {
                            if (window.reqresAnalytics && window.reqresAnalytics.legacyConsoleUsed) {
                                window.reqresAnalytics.legacyConsoleUsed({ endpoint: active.path, method: active.method, has_api_key: false });
                            }
                        } catch (e) {}

                        if (res.status === 429) {
                            var msg = 'You have used up the anonymous demo for today. Create a free account for a higher limit and your own API key.';
                            if (parsed && parsed.signup && parsed.signup.message) msg = parsed.signup.message;
                            elNudge.innerHTML = escapeHtml(msg) + ' <a href="https://app.reqres.in/sign-up?source=demo_rate_limit">Sign up free</a>';
                            elNudge.classList.remove('rp-hidden');
                            track('demo_rate_limited', { endpoint: active.id, path: active.path });
                        } else {
                            sentCount++;
                            try { localStorage.setItem('rr_demo_sent', String(sentCount)); } catch (e) {}
                            if (sentCount >= SIGNUP_THRESHOLD) { lockSend(); openModal(); track('demo_gate_reached', { sent_count: sentCount }); }
                        }
                    });
                }).catch(function (err) {
                    elStatus.className = 'rp-status-5';
                    elStatus.textContent = 'Network error';
                    elOut.textContent = '// ' + (err && err.message ? err.message : 'Request failed');
                    track('demo_request_error', { endpoint: active.id, message: (err && err.message) || 'unknown' });
                }).then(function () { busy(false); });
            });
        })();

    <script>
    (function() {
      const btns = document.querySelectorAll('#diagramToggle button');
      const tagline = document.getElementById('diagramTagline');
    
      // Desktop elements
      const infraNodes = document.querySelectorAll('#diagramDesktop .infra-node');
      const clientNodes = document.querySelectorAll('#diagramDesktop .client-node');
      const urlHub = document.getElementById('urlHub');
      const beforeLines = document.querySelectorAll('.before-line');
      const youLines = document.querySelectorAll('.you-line');
      const afterLines = document.querySelectorAll('.after-line');
      const afterDots = document.querySelectorAll('.after-dot');
      const afterPackets = document.querySelectorAll('.after-packet');
    
      // Mobile elements
      const mobileBefore = document.getElementById('mobileBefore');
      const mobileAfter = document.getElementById('mobileAfter');
    
      let currentPhase = 'before';
      let autoTimer = null;
      let userInteracted = false;
      let sectionVisible = false;
    
      const prefersReducedMotion = !!(window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches);
      const desktopSvg = document.querySelector('#diagramDesktop svg');
    
      const startAuto = () => {
        if (autoTimer || prefersReducedMotion || userInteracted || !sectionVisible || document.hidden) return;
        autoTimer = setInterval(() => { setPhase(currentPhase === 'before' ? 'with' : 'before'); }, 4500);
      };
      const stopAuto = () => {
        if (autoTimer) { clearInterval(autoTimer); autoTimer = null; }
      };
      const pauseSmil = () => { if (desktopSvg && desktopSvg.pauseAnimations) desktopSvg.pauseAnimations(); };
      const resumeSmil = () => { if (desktopSvg && desktopSvg.unpauseAnimations) desktopSvg.unpauseAnimations(); };
    
      // Store original desktop positions
      const origPos = {};
      infraNodes.forEach(n => { origPos[n.id] = { left: n.style.left, top: n.style.top, right: n.style.right }; });
    
      function setPhase(phase) {
        currentPhase = phase;
        btns.forEach(b => {
          const active = b.dataset.phase === phase;
          b.className = 'phase-btn text-[10px] uppercase tracking-[0.14em] font-semibold px-4 py-1.5 rounded-md border cursor-pointer '
            + (active ? 'border-emerald-400/25 bg-emerald-500/[0.06] text-emerald-300' : 'border-white/[0.08] bg-transparent text-white/50');
        });
    
        // ── Desktop ──
        if (phase === 'before') {
          infraNodes.forEach(n => {
            const o = origPos[n.id];
            if (o.left) n.style.left = o.left;
            if (o.top) n.style.top = o.top;
            if (o.right) n.style.right = o.right;
            n.style.opacity = '1';
            n.style.transform = '';
          });
          beforeLines.forEach(l => l.setAttribute('stroke-opacity', '0.04'));
          youLines.forEach(l => l.setAttribute('stroke-opacity', '0.03'));
          afterLines.forEach(l => l.setAttribute('stroke-opacity', '0'));
          afterDots.forEach(d => d.setAttribute('opacity', '0'));
          afterPackets.forEach(p => p.setAttribute('opacity', '0'));
          clientNodes.forEach(n => n.style.opacity = '0');
          urlHub.style.opacity = '0';
        } else {
          infraNodes.forEach(n => {
            n.style.opacity = '0';
            n.style.transform = 'scale(0.3)';
            n.style.left = '42%';
            n.style.top = '42%';
            n.style.right = 'auto';
          });
          beforeLines.forEach(l => l.setAttribute('stroke-opacity', '0'));
          youLines.forEach(l => l.setAttribute('stroke-opacity', '0'));
          afterLines.forEach(l => l.setAttribute('stroke-opacity', '0.15'));
          afterDots.forEach(d => d.setAttribute('opacity', '1'));
          afterPackets.forEach(p => p.setAttribute('opacity', '0.8'));
          clientNodes.forEach((n, i) => { setTimeout(() => { n.style.opacity = '1'; }, 200 + i * 120); });
          urlHub.style.opacity = '1';
        }
    
        // ── Mobile ──
        if (phase === 'before') {
          mobileBefore.style.display = '';
          mobileAfter.style.display = 'none';
        } else {
          mobileBefore.style.display = 'none';
          mobileAfter.style.display = '';
        }
    
        // ── Tagline ──
        if (phase === 'before') {
          tagline.innerHTML = '<span class="text-white/65 font-medium">80% of backends</span> are CRUD + auth.<br>That\'s a lot of infra you\'re maintaining for nothing.';
        } else {
          tagline.innerHTML = '<span class="text-white/65 font-medium">Database, auth, CRUD, hosting</span> - handled.<br>Just send requests.';
        }
      }
    
      btns.forEach(b => b.addEventListener('click', () => {
        userInteracted = true;
        stopAuto();
        setPhase(b.dataset.phase);
      }));
      setPhase('before');
    
      const section = document.getElementById('how');
      if (section && 'IntersectionObserver' in window) {
        const io = new IntersectionObserver((entries) => {
          entries.forEach(entry => {
            sectionVisible = entry.isIntersecting;
            if (sectionVisible) { resumeSmil(); startAuto(); }
            else { pauseSmil(); stopAuto(); }
          });
        }, { threshold: 0.1 });
        io.observe(section);
      } else {
        sectionVisible = true;
        startAuto();
      }
    
      document.addEventListener('visibilitychange', () => {
        if (document.hidden) { pauseSmil(); stopAuto(); }
        else if (sectionVisible) { resumeSmil(); startAuto(); }
      });
    })();

        let id = 'rec_';
        for (let i = 0; i < 8; i++) {
          id += chars.charAt(Math.floor(Math.random() * chars.length));
        }
        return id;
      }

        return new Promise(resolve => setTimeout(resolve, ms));

      function highlightJSON(obj) {
        if (obj === null) return '<span class="json-null">null</span>';
        if (obj === undefined) return '';
    
        const json = JSON.stringify(obj, null, 2);
        return highlightJSONString(json);
      }
    
      // JSON syntax highlighting (for raw strings)
      function highlightJSONString(str) {
        if (!str) return '';
        return str
          .replace(/&/g, '&amp;')
          .replace(/</g, '&lt;')
          .replace(/>/g, '&gt;')
          .replace(/"([^"]+)":/g, '<span class="json-key">"$1"</span>:')
          .replace(/: "([^"]*)"(,?)/g, ': <span class="json-string">"$1"</span>$2')
          .replace(/: (-?\d+\.?\d*)(,?)/g, ': <span class="json-number">$1</span>$2')
          .replace(/: (true|false)(,?)/g, ': <span class="json-boolean">$1</span>$2')
          .replace(/: (null)(,?)/g, ': <span class="json-null">$1</span>$2')
          .replace(/([{}\[\]])/g, '<span class="json-bracket">$1</span>');
      }

        line4.innerHTML = '<span class="text-emerald-400">> Database, API, auth, logs - live.</span>';
        bootLinesContainer.appendChild(line4);
        await sleep(300);
    
        // Empty line
        const emptyLine = document.createElement('div');
        emptyLine.innerHTML = '&nbsp;';
        bootLinesContainer.appendChild(emptyLine);
    
        const line5 = document.createElement('div');
        line5.className = 'boot-line';
        line5.innerHTML = '<span class="text-white/60">> Try it <span class="arrow-bounce">\u2193</span></span>';
        bootLinesContainer.appendChild(line5);
        bootLinesContainer.appendChild(cursor);
    
        // Pause before transitioning to explorer
        await sleep(1500);
        transitionToExplorer();
      }
    
      function showFinalBootState() {
        const config = getConfig();
        const bootLinesContainer = elements.bootLines;
    
        // Show struck checklist
        const line1 = document.createElement('div');
        line1.style.opacity = '1';
        line1.innerHTML = '<span class="text-white/70">> Setting up your backend the usual way...</span>';
        bootLinesContainer.appendChild(line1);
    
        checklistItems.forEach(item => {
          const el = document.createElement('div');
          el.className = 'checklist-item struck';
          el.style.opacity = '1';
          el.innerHTML = `<span class="checkbox">\u25A1</span><span class="text">${item}</span>`;
          bootLinesContainer.appendChild(el);
        });
    
        const line2 = document.createElement('div');
        line2.style.opacity = '1';
        line2.innerHTML = '<span class="text-white/70">> Actually... </span><span class="text-emerald-400">skip all that.</span>';
        bootLinesContainer.appendChild(line2);
    
        const line3 = document.createElement('div');
        line3.style.opacity = '1';
        line3.innerHTML = '<span class="text-emerald-400">> reqres.in/api/' + config.projectName + ' \u2713</span>';
        bootLinesContainer.appendChild(line3);
    
        const line4 = document.createElement('div');
        line4.style.opacity = '1';
        line4.innerHTML = '<span class="text-emerald-400">> Database, API, auth, logs - live.</span>';
        bootLinesContainer.appendChild(line4);
    
        const emptyLine = document.createElement('div');
        emptyLine.innerHTML = '&nbsp;';
        bootLinesContainer.appendChild(emptyLine);
    
        const line5 = document.createElement('div');
        line5.style.opacity = '1';
        line5.innerHTML = '<span class="text-white/60">> Try it \u2193</span>';
        bootLinesContainer.appendChild(line5);
    
        elements.bootCursor.style.display = 'none';
      }
    
      function transitionToExplorer() {
        state.bootComplete = true;
        state.phase = 'explorer';
    
        elements.bootSequence.style.display = 'none';
        elements.apiExplorer.classList.remove('explorer-dimmed');
        elements.apiExplorer.classList.add('explorer-active');
    
        // Set default body for current audience
        const config = getConfig();
        elements.requestBody.value = JSON.stringify(config.defaultBody, null, 2);
        updateRequestBodyHighlight();
    
        // Update endpoint
        updateEndpoint();
    
        // Set initial method
        setMethod('POST');
      }
    
      // Update endpoint display based on current audience and method
      function updateEndpoint() {
        const config = getConfig();
        const basePath = 'https://reqres.in/api/' + config.projectName + '/';
        elements.endpointBase.textContent = basePath;
    
        if (state.method === 'PUT' || state.method === 'DELETE') {
          if (state.records.length > 0) {
            elements.endpointPath.textContent = 'collections/' + config.collectionName + '/records/' + state.records[0].id;
          } else {
            elements.endpointPath.textContent = 'collections/' + config.collectionName + '/records/:id';
          }
        } else {
          elements.endpointPath.textContent = 'collections/' + config.collectionName + '/records';
        }
      }
    
      // Method switching
      function setMethod(method) {
        state.method = method;
    
        // Update pills
        document.querySelectorAll('.method-pill').forEach(pill => {
          pill.classList.toggle('active', pill.dataset.method === method);
          pill.classList.remove('pulse-get');
        });
    
        // Show/hide body editor
        const showBody = method === 'POST' || method === 'PUT';
        elements.bodyEditorContainer.style.display = showBody ? 'block' : 'none';
    
        // Update endpoint
        updateEndpoint();
    
        // Pre-fill PUT body
        if (method === 'PUT' && state.records.length > 0) {
          const record = state.records[0];
          if (state.audience === 'building') {
            elements.requestBody.value = JSON.stringify({
              title: record.title,
              completed: true
            }, null, 2);
          } else {
            elements.requestBody.value = JSON.stringify({
              name: record.name,
              email: record.email,
              role: 'superadmin'
            }, null, 2);
          }
          updateRequestBodyHighlight();
        } else if (method === 'POST') {
          const config = getConfig();
          elements.requestBody.value = JSON.stringify(config.defaultBody, null, 2);
          updateRequestBodyHighlight();
        }
    
        // Hide nudge prompt when switching to GET
        if (method === 'GET') {
          elements.nudgePrompt.classList.add('hidden');
        }
      }
    
      // API handlers with realistic response times
      const apiHandlers = {
        GET: () => {
          const delay = 8 + Math.random() * 17; // 8-25ms
          return {
            status: 200,
            statusText: '200 OK',
            statusClass: 'status-success',
            delay,
            body: {
              data: state.records,
              total: state.records.length,
              page: 1,
              per_page: 20
            }
          };
        },
    
        POST: (body) => {
          const delay = 18 + Math.random() * 27; // 18-45ms
          const record = {
            id: generateId(),
            ...body,
            createdAt: getTimestamp(),
            updatedAt: getTimestamp()
          };
          state.records.push(record);
          return {
            status: 201,
            statusText: '201 Created',
            statusClass: 'status-success',
            delay,
            body: record,
            showNudge: state.requestCount === 0
          };
        },
    
        PUT: (body) => {
          const delay = 15 + Math.random() * 25; // 15-40ms
          if (state.records.length === 0) {
            return {
              status: 404,
              statusText: '404 Not Found',
              statusClass: 'status-error',
              delay,
              body: { error: 'No record found to update' }
            };
          }
          const record = state.records[0];
          Object.assign(record, body, { updatedAt: getTimestamp() });
          return {
            status: 200,
            statusText: '200 OK',
            statusClass: 'status-success',
            delay,
            body: record
          };
        },
    
        DELETE: () => {
          const delay = 8 + Math.random() * 10; // 8-18ms
          if (state.records.length === 0) {
            return {
              status: 404,
              statusText: '404 Not Found',
              statusClass: 'status-error',
              delay,
              body: { error: 'No record found to delete' }
            };
          }
          state.records.shift();
          return {
            status: 204,
            statusText: '204 No Content',
            statusClass: 'status-neutral',
            delay,
            body: null
          };
        }
      };
    
      // Send request
      async function sendRequest() {
        if (state.isLoading) return;
    
        state.isLoading = true;
        state.hasInteracted = true;
    
        // Button animation
        elements.sendBtn.classList.add('send-click');
        setTimeout(() => elements.sendBtn.classList.remove('send-click'), 100);
    
        // Show loading dots
        elements.sendBtnText.textContent = 'Sending';
        elements.sendBtnArrow.classList.add('hidden');
        elements.sendBtnDots.classList.remove('hidden');
    
        // Parse body if needed
        let body = null;
        if (state.method === 'POST' || state.method === 'PUT') {
          try {
            body = JSON.parse(elements.requestBody.value);
          } catch (e) {
            showResponse({
              status: 400,
              statusText: '400 Bad Request',
              statusClass: 'status-error',
              delay: 5,
              body: { error: 'Invalid JSON in request body' }
            });
            resetSendButton();
            return;
          }
        }
    
        const handler = apiHandlers[state.method];
        const response = handler(body);
    
        // Simulate network delay (add 150ms base for realism)
        await sleep(response.delay + 150);
    
        showResponse(response);
        resetSendButton();
    
        state.requestCount++;
    
        // Show CTA after 2+ requests
        if (state.requestCount >= 2 && !state.ctaShown) {
          showCTA();
        }
      }
    
      function resetSendButton() {
        state.isLoading = false;
        elements.sendBtnText.textContent = 'Send';
        elements.sendBtnArrow.classList.remove('hidden');
        elements.sendBtnDots.classList.add('hidden');
      }
    
      function showResponse(response) {
        // Status badge
        elements.responseStatus.innerHTML = `
          <span class="status-badge ${response.statusClass}">${response.statusText}</span>
          <span class="text-[11px] text-white/40 terminal-font">${Math.round(response.delay)}ms</span>
        `;
    
        // Response body
        if (response.body === null) {
          elements.responseBody.innerHTML = '<span class="text-white/40">// No content</span>';
        } else {
          elements.responseBody.innerHTML = highlightJSON(response.body);
        }
    
        // Show nudge after first POST
        if (response.showNudge) {
          // Wait 1 second then show nudge
          setTimeout(() => {
            const config = getConfig();
            elements.nudgePrompt.querySelector('#nudgeText').textContent = config.nudgeText;
            elements.nudgePrompt.classList.remove('hidden');
    
            // Add pulse animation to GET pill
            const getPill = document.querySelector('.method-pill[data-method="GET"]');
            if (getPill) {
              getPill.classList.add('pulse-get');
            }
          }, 1000);
        }
      }
    
      function showCTA() {
        state.ctaShown = true;
        elements.ctaBanner.classList.remove('hidden');
        elements.ctaBanner.classList.add('cta-banner-show');
      }
    
      // Audience toggle
      function setAudience(newAudience) {
        if (newAudience === state.audience) return;
    
        state.audience = newAudience;
    
        // Update toggle buttons
        document.querySelectorAll('.audience-btn').forEach(btn => {
          btn.classList.toggle('active', btn.dataset.audience === newAudience);
        });
    
        const config = getConfig();
    
        // Fade out tagline, update, fade in
        elements.audienceTagline.style.opacity = '0';
        setTimeout(() => {
          elements.audienceTagline.textContent = config.tagline;
          elements.audienceTagline.style.opacity = '1';
        }, 150);
    
        // If boot is complete, update terminal content
        if (state.bootComplete) {
          // Reset state
          state.records = [];
          state.requestCount = 0;
          state.ctaShown = false;
          elements.ctaBanner.classList.add('hidden');
          elements.ctaBanner.classList.remove('cta-banner-show');
          elements.nudgePrompt.classList.add('hidden');
    
          // Remove pulse from GET pill
          const getPill = document.querySelector('.method-pill[data-method="GET"]');
          if (getPill) getPill.classList.remove('pulse-get');
    
          // Smooth content transition if user has interacted
          if (state.hasInteracted) {
            // Fade out
            elements.requestBody.style.opacity = '0';
            elements.requestBodyHighlight.style.opacity = '0';
            elements.responseBody.style.opacity = '0';
    
            setTimeout(() => {
              // Update content
              elements.requestBody.value = JSON.stringify(config.defaultBody, null, 2);
              updateRequestBodyHighlight();
              updateEndpoint();
              elements.responseBody.innerHTML = '<span class="text-white/60">// Response will appear here</span>';
              elements.responseStatus.innerHTML = '';
    
              // Reset to POST
              setMethod('POST');
    
              // Fade in
              elements.requestBody.style.opacity = '1';
              elements.requestBodyHighlight.style.opacity = '1';
              elements.responseBody.style.opacity = '1';
            }, 150);
          } else {
            // Instant update if no interaction yet
            elements.requestBody.value = JSON.stringify(config.defaultBody, null, 2);
            updateRequestBodyHighlight();
            updateEndpoint();
            setMethod('POST');
          }
        }
      }
    
      // Initialize
      function init() {
        // Cache DOM elements
        elements = {
          terminalWidget: document.getElementById('terminalWidget'),
          bootSequence: document.getElementById('bootSequence'),
          bootLines: document.getElementById('bootLines'),
          bootCursor: document.getElementById('bootCursor'),
          apiExplorer: document.getElementById('apiExplorer'),
          methodPills: document.getElementById('methodPills'),
          requestBody: document.getElementById('requestBody'),
          requestBodyHighlight: document.getElementById('requestBodyHighlight'),
          bodyEditorContainer: document.getElementById('bodyEditorContainer'),
          endpointBase: document.getElementById('endpointBase'),
          endpointPath: document.getElementById('endpointPath'),
          sendBtn: document.getElementById('sendBtn'),
          sendBtnText: document.getElementById('sendBtnText'),
          sendBtnArrow: document.getElementById('sendBtnArrow'),
          sendBtnDots: document.getElementById('sendBtnDots'),
          responseStatus: document.getElementById('responseStatus'),
          responseBody: document.getElementById('responseBody'),
          nudgePrompt: document.getElementById('nudgePrompt'),
          ctaBanner: document.getElementById('ctaBanner'),
          audienceTagline: document.getElementById('audienceTagline')
        };
    
        if (!elements.terminalWidget) return;
    
        // Set initial content
        const config = getConfig();
        elements.requestBody.value = JSON.stringify(config.defaultBody, null, 2);
        updateRequestBodyHighlight();
    
        // Method pill clicks
        elements.methodPills.addEventListener('click', (e) => {
          const pill = e.target.closest('.method-pill');
          if (pill && !state.isLoading && state.bootComplete) {
            setMethod(pill.dataset.method);
          }
        });
    
        // Send button click
        elements.sendBtn.addEventListener('click', () => {
          if (state.bootComplete) sendRequest();
        });
    
        // Keyboard: Enter to send when focused on textarea
        elements.requestBody.addEventListener('keydown', (e) => {
          if (e.key === 'Enter' && (e.metaKey || e.ctrlKey) && state.bootComplete) {
            e.preventDefault();
            sendRequest();
          }
        });
    
        // Sync syntax highlighting with textarea input
        elements.requestBody.addEventListener('input', updateRequestBodyHighlight);
        elements.requestBody.addEventListener('scroll', () => {
          if (elements.requestBodyHighlight) {
            elements.requestBodyHighlight.scrollTop = elements.requestBody.scrollTop;
            elements.requestBodyHighlight.scrollLeft = elements.requestBody.scrollLeft;
          }
        });
    
        // Nudge prompt click - switch to GET
        elements.nudgePrompt.addEventListener('click', () => {
          setMethod('GET');
        });
    
        // Audience toggle
        document.getElementById('audienceToggle').addEventListener('click', (e) => {
          const btn = e.target.closest('.audience-btn');
          if (btn) {
            setAudience(btn.dataset.audience);
          }
        });
    
        // IntersectionObserver for boot trigger
        const observer = new IntersectionObserver((entries) => {
          entries.forEach(entry => {
            if (entry.isIntersecting && !state.bootComplete) {
              observer.unobserve(entry.target);
              startBootSequence();
            }
          });
        }, { threshold: 0.3 });
    
        observer.observe(elements.terminalWidget);
      }
    
      // Run on DOM ready
      if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', init);
      } else {
        init();
      }
    })();
    </script>

            (function () {
                var body = document.getElementById('abTermBody')
                var caret = document.getElementById('abCaret')
                if (!body) return
                var reduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches
                var lines = [
                    { t: 'cmd', txt: 'curl "reqres.in/agent/v1/users?limit=1"' },
                    { t: 'dim', txt: 'run 1 \u00b7 sha256 dfc806a7760d\u2026' },
                    { t: 'cmd', txt: 'curl "reqres.in/agent/v1/users?limit=1"' },
                    { t: 'dim', txt: 'run 2 \u00b7 sha256 dfc806a7760d\u2026' },
                    { t: 'pass', txt: '\u2713 byte-identical across runs' },
                ]
                function render(upTo, partial) {
                    var html = ''
                    for (var i = 0; i < upTo; i++) {
                        var l = lines[i]
                        if (l.t === 'cmd') html += '<div class="text-white"><span class="text-amber-400">$ </span>' + l.txt + '</div>'
                        if (l.t === 'dim') html += '<div class="text-white/40">' + l.txt + '</div>'
                        if (l.t === 'pass') html += '<div class="text-emerald-400 font-semibold mt-2">' + l.txt + '</div>'
                    }
                    if (partial !== undefined) html += '<div class="text-white"><span class="text-amber-400">$ </span>' + partial + '</div>'
                    body.innerHTML = html
                    if (caret && partial !== undefined) body.appendChild(caret)
                }
                function play() {
                    var step = 0
                    ;(function next() {
                        if (step >= lines.length) { if (caret) caret.style.display = 'none'; return }
                        var l = lines[step]
                        if (l.t === 'cmd') {
                            var i = 0
                            ;(function type() {
                                if (i <= l.txt.length) { render(step, l.txt.slice(0, i)); i++; setTimeout(type, 20) }
                                else { step++; setTimeout(next, 260) }
                            })()
                        } else { step++; render(step); setTimeout(next, l.t === 'pass' ? 0 : 420) }
                    })()
                }
                if (reduced) { render(lines.length); if (caret) caret.style.display = 'none'; return }
                var fired = false
                var obs = new IntersectionObserver(function (entries) {
                    entries.forEach(function (e) {
                        if (e.isIntersecting && !fired) { fired = true; obs.disconnect(); setTimeout(play, 350) }
                    })
                }, { threshold: 0.35 })
                obs.observe(body)
            })()
        </script>

    (function() {
      const cards = document.querySelectorAll('#compareCards .compare-card');

      cards.forEach(card => {
        // Make cards keyboard-focusable and announce as buttons
        card.setAttribute('role', 'button');
        card.setAttribute('tabindex', '0');
        card.setAttribute('aria-expanded', 'false');

        function toggle(e) {
          // Don't toggle if clicking the comparison link
          if (e.target.closest('a')) return;

          const answer = card.querySelector('.compare-answer');
          const isActive = card.classList.contains('active');

          // Close all
          cards.forEach(c => {
            c.classList.remove('active');
            c.querySelector('.compare-answer').classList.remove('open');
            c.setAttribute('aria-expanded', 'false');
          });

          // Open clicked (if it wasn't already open)
          if (!isActive) {
            card.classList.add('active');
            answer.classList.add('open');
            card.setAttribute('aria-expanded', 'true');
          }
        }

        card.addEventListener('click', toggle);
        card.addEventListener('keydown', (e) => {
          if (e.key === 'Enter' || e.key === ' ') {
            e.preventDefault();
            toggle(e);
          }
        });
      });
    })();
    </script>
